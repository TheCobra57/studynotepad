    51→export interface BlockEditorHandle {

    64→  /** swap the src of an existing image block (whiteboard annotation flow) */

   142→ * Every block is rendered (headings, styled lists, live KaTeX equations…).
   143→ * Clicking a block swaps that single block into an in-place editor, so raw
   144→ * markdown/LaTeX is only ever visible for the block you are editing.
   145→ * Typing "/" opens the quick-insert command menu.
   146→ */

   148→  function BlockEditor({ noteId, content, onChange, onImageAction, onAnnotateImage }, ref) {

   151→    const [slash, setSlash] = useState<SlashState | null>(null)
   152→    const [dragIndex, setDragIndex] = useState<number | null>(null)

   154→    const [selectedImageId, setSelectedImageId] = useState<string | null>(null)
   155→    const [imgResize, setImgResize] = useState<{ id: string; w: number } | null>(null)
   156→    const [fileDragOver, setFileDragOver] = useState(false)
   157→    /** heading block briefly highlighted after a ToC link jump */
   158→    const [flashId, setFlashId] = useState<string | null>(null)
   159→
   160→    const taRefs = useRef(new Map<string, HTMLTextAreaElement>())
   161→    const blocksRef = useRef(blocks)
   162→    const serializedRef = useRef(serializeBlocks(blocks))
   163→    const activeIdRef = useRef<string | null>(null)
   164→    const slashRef = useRef<SlashState | null>(null)
   165→    const onChangeRef = useRef(onChange)
   166→    const debounceRef = useRef<number | undefined>(undefined)
   167→    const blurTimerRef = useRef<number | undefined>(undefined)
   168→    const pendingImageRef = useRef<{ replaceId?: string; afterId?: string } | null>(null)
   169→    const onImageActionRef = useRef(onImageAction)
   170→    const onAnnotateImageRef = useRef(onAnnotateImage)
   171→    const flashTimerRef = useRef<number | undefined>(undefined)

   173→    // keep refs in sync after every render (lint-safe: no writes during render)

   175→      blocksRef.current = blocks
   176→      activeIdRef.current = activeId
   177→      slashRef.current = slash

   179→      onImageActionRef.current = onImageAction
   180→      onAnnotateImageRef.current = onAnnotateImage

   183→    // ------------------------------------------------------------------
   184→    // persistence — debounced markdown sync + flush on unmount / leave
   185→    // ------------------------------------------------------------------
   186→
   187→    const commit = (next: Block[]) => {
   188→      blocksRef.current = next
   189→      setBlocks(next)
   190→      const md = serializeBlocks(next)
   191→      serializedRef.current = md
   192→      if (debounceRef.current !== undefined) window.clearTimeout(debounceRef.current)
   193→      debounceRef.current = window.setTimeout(() => {
   194→        debounceRef.current = undefined
   195→        onChangeRef.current(md)
   196→      }, 600)
   197→    }
   198→
   199→    const flush = () => {
   200→      if (debounceRef.current === undefined) return
   201→      window.clearTimeout(debounceRef.current)
   202→      debounceRef.current = undefined
   203→      onChangeRef.current(serializeBlocks(blocksRef.current))
   204→    }
   205→
   206→    useEffect(() => {
   207→      const onLeave = () => flush()
   208→      window.addEventListener('beforeunload', onLeave)
   209→      return () => {
   210→        window.removeEventListener('beforeunload', onLeave)
   211→        flush()
   212→      }
   213→    }, [])
   214→
   215→    // external content change while the same note stays open (e.g. import)
   216→    useEffect(() => {
   217→      if (content === serializedRef.current) return
   218→      const t = window.setTimeout(() => {
   219→        if (activeIdRef.current) return // never clobber an in-progress edit
   220→        if (onChangeRef.current === undefined) return
   221→        const parsed = parseMarkdownToBlocks(content)
   222→        serializedRef.current = serializeBlocks(parsed)
   223→        blocksRef.current = parsed
   224→        setBlocks(parsed)
   225→        setActiveId(null)
   226→        setSlash(null)
   227→      }, 0)
   228→      return () => window.clearTimeout(t)
   229→    }, [content, noteId])
   230→

   242→          return

   245→        setSlash(null)
   246→      }

   248→      return () => document.removeEventListener('pointerdown', onPointerDown)
   249→    }, [activeId])
   250→

   252→    useEffect(() => {
   253→      if (!selectedImageId) return

   255→        const t = e.target as HTMLElement | null
   256→        if (!t) return
   257→        if (t.closest('[data-image-ui]')) return

   259→      }
   260→      document.addEventListener('pointerdown', onPointerDown)
   261→      return () => document.removeEventListener('pointerdown', onPointerDown)
   262→    }, [selectedImageId])
   263→
   264→    useEffect(
   265→      () => () => {
   266→        if (blurTimerRef.current !== undefined) window.clearTimeout(blurTimerRef.current)
   267→        if (flashTimerRef.current !== undefined) window.clearTimeout(flashTimerRef.current)
   268→      },
   269→      [],
   270→    )

   272→    // ------------------------------------------------------------------
   273→    // focus helpers
   274→    // ------------------------------------------------------------------
   275→
   276→    const focusBlock = (
   277→      id: string,
   278→      caret: 'start' | 'end' | number = 'end',
   279→      selEnd?: number,
   280→    ) => {
   281→      requestAnimationFrame(() => {
   282→        const ta = taRefs.current.get(id)
   283→        if (!ta) return
   284→        ta.focus()

   286→        let pos: number
   287→        if (caret === 'start') pos = 0
   288→        else if (caret === 'end') pos = len
   289→        else pos = caret < 0 ? Math.max(0, len + caret) : Math.min(caret, len)
   290→        ta.setSelectionRange(pos, selEnd !== undefined ? Math.min(selEnd, len) : pos)
   291→        autoResize(ta)
   292→      })
   293→    }
   294→
   295→    const registerTa = (id: string) => (el: HTMLTextAreaElement | null) => {

   297→        taRefs.current.set(id, el)
   298→        autoResize(el)
   299→      } else {
   300→        taRefs.current.delete(id)
   301→      }
   302→    }
   303→
   304→    const cancelBlurDeactivate = () => {
   305→      if (blurTimerRef.current !== undefined) {
   306→        window.clearTimeout(blurTimerRef.current)
   307→        blurTimerRef.current = undefined
   308→      }
   309→    }
   310→
   311→    const handleFocus = (id: string) => {
   312→      cancelBlurDeactivate()

   314→    }
   315→
   316→    const handleBlurDeactivate = (id: string, relatedTarget: EventTarget | null) => {
   317→      cancelBlurDeactivate()

   319→      // menu — the toolbar needs to act on the block the user was just editing.
   320→      const rt = relatedTarget as HTMLElement | null
   321→      if (rt && rt.closest('[role="toolbar"], [role="menu"]')) return

   323→        blurTimerRef.current = undefined
   324→        setActiveId((cur) => (cur === id ? null : cur))
   325→        setSlash((cur) => (cur && cur.id === id ? null : cur))
   326→      }, 140)
   327→    }
   328→
   329→    const activate = (id: string, caret: 'start' | 'end' | number = 'end') => {
   330→      cancelBlurDeactivate()
   331→      setActiveId(id)
   332→      focusBlock(id, caret)
   333→    }
   334→
   335→    // ------------------------------------------------------------------
   336→    // block operations
   337→    // ------------------------------------------------------------------
   338→
   339→    const setBlockText = (id: string, text: string) => {
   340→      commit(blocksRef.current.map((b) => (b.id === id ? { ...b, text } : b)))
   341→    }
   342→
   343→    const transformBlock = (id: string, type: BlockType, text?: string) => {
   344→      commit(
   345→        blocksRef.current.map((b) => {
   346→          if (b.id !== id) return b
   347→          const next: Block = { ...b, type, text: text !== undefined ? text : b.text }
   348→          if (type === 'todo' && next.checked === undefined) next.checked = false
   349→          if (type === 'code' && next.lang === undefined) next.lang = ''
   350→          if (type !== 'todo') delete next.checked
   351→          if (type !== 'code') delete next.lang
   352→          if (type !== 'image') {
   353→            delete next.src
   354→            delete next.width
   355→          }
   356→          return next
   357→        }),
   358→      )
   359→    }
   360→
   361→    const insertBlockAfter = (id: string | null, block: Block) => {
   362→      const arr = [...blocksRef.current]
   363→      const idx = id ? arr.findIndex((b) => b.id === id) : arr.length - 1
   364→      const at = idx === -1 ? arr.length : idx + 1
   365→      arr.splice(at, 0, block)
   366→      commit(arr)
   367→      return block
   368→    }
   369→
   370→    const removeBlock = (id: string) => {
   371→      const arr = blocksRef.current.filter((b) => b.id !== id)
   372→      if (arr.length === 0) arr.push(createBlock('p'))

   374→    }
   375→
   376→    // keyboard: Escape deselects, Delete/Backspace removes the selected image
   377→    useEffect(() => {
   378→      if (!selectedImageId) return
   379→      const onKeyDown = (e: KeyboardEvent) => {
   380→        const ae = document.activeElement
   381→        const typing =
   382→          ae instanceof HTMLElement &&
   383→          (ae.tagName === 'TEXTAREA' || ae.tagName === 'INPUT' || ae.isContentEditable)
   384→        if (typing) return
   385→        if (e.key === 'Escape') {

   390→          setSelectedImageId(null)
   391→        }
   392→      }

   394→      return () => document.removeEventListener('keydown', onKeyDown)

   399→    }

   405→      const [moved] = arr.splice(idx, 1)

   408→    }

   432→     * Place processed images into the document. Honours a pending anchor set
   433→     * by a slash command (replace the empty "…/image" paragraph, or insert
   434→     * after the block that typed it); otherwise replaces an empty active
   435→     * paragraph or inserts after the active block / at the end.
   436→     */
   437→    const insertImagesInternal = (srcs: string[]) => {
   438→      if (!srcs.length) return
   439→      const arr = [...blocksRef.current]
   440→      let replaceId = pendingImageRef.current?.replaceId
   441→      const afterPending = pendingImageRef.current?.afterId
   442→      pendingImageRef.current = null
   443→
   444→      if (!replaceId) {
   445→        const active = activeIdRef.current
   446→          ? arr.find((b) => b.id === activeIdRef.current)
   447→          : null
   448→        if (active && active.type === 'p' && active.text.trim() === '') {
   449→          replaceId = active.id
   450→        }
   451→      }
   452→
   453→      const mkImage = (src: string): Block => ({
   454→        id: newBlockId(),
   455→        type: 'image',
   456→        text: '',
   457→        src,
   458→        width: 100,
   459→      })
   460→
   461→      let first: Block
   462→      if (replaceId) {
   463→        const i = arr.findIndex((b) => b.id === replaceId)
   464→        if (i >= 0) {
   465→          first = mkImage(srcs[0])
   466→          arr[i] = first
   467→          let at = i + 1
   468→          for (const s of srcs.slice(1)) arr.splice(at++, 0, mkImage(s))
   469→          commit(arr)
   470→          setActiveId(null)
   471→          setSlash(null)
   472→          setSelectedImageId(first.id)
   473→          return
   474→        }
   475→      }
   476→      const anchorId = afterPending ?? activeIdRef.current
   477→      let idx = anchorId ? arr.findIndex((b) => b.id === anchorId) : -1
   478→      if (idx === -1) idx = arr.length - 1
   479→      first = mkImage(srcs[0])
   480→      arr.splice(idx + 1, 0, first)
   481→      let at = idx + 2
   482→      for (const s of srcs.slice(1)) arr.splice(at++, 0, mkImage(s))
   483→      commit(arr)
   484→      setActiveId(null)
   485→      setSlash(null)
   486→      setSelectedImageId(first.id)
   487→    }

   537→    /** presentational ToC block — entries are clickable, content auto-updates */

   622→      autoResize(e.target)

   630→          setSlash((prev) => ({
   631→            id: block.id,

   643→    const applyCommand = (cmd: SlashCommand) => {
   644→      const current = slashRef.current
   645→      const targetId =
   646→        current?.id ?? activeIdRef.current ?? blocksRef.current[blocksRef.current.length - 1]?.id
   647→      if (!targetId) return
   648→      const target = blocksRef.current.find((b) => b.id === targetId)
   649→      if (!target) return
   650→
   651→      // strip the typed "/query" that opened the menu
   652→      let stripped = target.text
   653→      if (current && current.id === targetId) {
   654→        const ta = taRefs.current.get(targetId)
   655→        const caret = ta ? (ta.selectionStart ?? target.text.length) : target.text.length
   656→        stripped = target.text.slice(0, current.start) + target.text.slice(caret)
   657→      }
   658→      setSlash(null)
   659→
   660→      // ---- inline insertions stay inside the current text block ----
   661→      if (cmd.inline) {
   662→        const at = current ? current.start : stripped.length
   663→        const next = stripped.slice(0, at) + cmd.inline + stripped.slice(at)
   664→        setBlockText(targetId, next)
   665→        cancelBlurDeactivate()
   666→        setActiveId(targetId)
   667→        focusBlock(targetId, at + cmd.inline.length)
   668→        return
   669→      }
   670→
   671→      // ---- image flows are orchestrated by the editor pane ----
   672→      if (cmd.imageAction) {
   673→        if (stripped !== target.text) setBlockText(targetId, stripped)
   674→        // an empty paragraph that only served as the "/" trigger is replaced
   675→        // by the incoming image; otherwise the image lands right below
   676→        if (target.type === 'p' && stripped.trim() === '') {
   677→          pendingImageRef.current = { replaceId: targetId }
   678→        } else {
   679→          pendingImageRef.current = { afterId: targetId }
   680→        }
   681→        setActiveId(null)
   682→        onImageActionRef.current?.(cmd.imageAction)
   683→        return
   684→      }
   685→
   686→      if (!cmd.blockType) return
   687→      const type = cmd.blockType
   688→
   689→      // ---- divider: never replaces text, inserts a line ----
   690→      if (type === 'divider') {
   691→        if (stripped.trim() === '' && target.type === 'p') {
   692→          const idx = blocksRef.current.findIndex((b) => b.id === targetId)
   693→          const arr = [...blocksRef.current]
   694→          arr[idx] = createBlock('divider')
   695→          const fresh = createBlock('p')
   696→          arr.splice(idx + 1, 0, fresh)
   697→          commit(arr)
   698→          activate(fresh.id, 'start')
   699→        } else {
   700→          if (stripped !== target.text) setBlockText(targetId, stripped)
   701→          insertBlockAfter(targetId, createBlock('divider'))
   702→          activate(targetId, 'end')
   703→        }
   704→        return
   705→      }
   706→
   707→      // ---- table of contents: same flow as the divider (block + fresh line) ----
   708→      if (type === 'toc') {
   709→        if (stripped.trim() === '' && target.type === 'p') {
   710→          const idx = blocksRef.current.findIndex((b) => b.id === targetId)
   711→          const arr = [...blocksRef.current]
   712→          arr[idx] = createBlock('toc')
   713→          const fresh = createBlock('p')
   714→          arr.splice(idx + 1, 0, fresh)
   715→          commit(arr)
   716→          activate(fresh.id, 'start')
   717→        } else {
   718→          if (stripped !== target.text) setBlockText(targetId, stripped)
   719→          insertBlockAfter(targetId, createBlock('toc'))
   720→          activate(targetId, 'end')
   721→        }
   722→        return
   723→      }
   724→
   725→      // ---- transform the empty paragraph in place, else insert below ----
   726→      if (target.type === 'p' && stripped.trim() === '') {
   727→        transformBlock(targetId, type, cmd.blockText ?? '')
   728→        cancelBlurDeactivate()
   729→        setActiveId(targetId)
   730→        focusBlock(targetId, cmd.caretFromEnd ?? (cmd.blockText ? 'end' : 'start'))
   731→        return
   732→      }
   733→
   734→      if (stripped !== target.text) setBlockText(targetId, stripped)
   735→      const nb = insertBlockAfter(targetId, createBlock(type, cmd.blockText ?? ''))
   736→      cancelBlurDeactivate()
   737→      setActiveId(nb.id)
   738→      focusBlock(nb.id, cmd.caretFromEnd ?? (cmd.blockText ? 'end' : 'start'))
   739→    }

   763→    const wrapSelection = (prefix: string, suffix: string, placeholder: string): boolean => {
   764→      const id = activeIdRef.current
   765→      if (!id) return false

   771→      const end = ta.selectionEnd ?? start

   779→    const handleTextKeyDown = (
   780→      e: ReactKeyboardEvent<HTMLTextAreaElement>,
   781→      block: Block,
   782→      idx: number,
   783→    ) => {
   784→      const ta = e.currentTarget
   785→      const { selectionStart: s, selectionEnd: en, value } = ta
   786→
   787→      // slash menu navigation
   788→      if (slash && slash.id === block.id && slashCommands.length > 0) {
   789→        if (e.key === 'ArrowDown') {
   790→          e.preventDefault()
   791→          setSlash((p) => (p ? { ...p, index: (p.index + 1) % slashCommands.length } : p))
   792→          return
   793→        }
   794→        if (e.key === 'ArrowUp') {
   795→          e.preventDefault()
   796→          setSlash((p) =>
   797→            p ? { ...p, index: (p.index - 1 + slashCommands.length) % slashCommands.length } : p,
   798→          )
   799→          return
   800→        }
   801→        if (e.key === 'Enter' || e.key === 'Tab') {
   802→          e.preventDefault()
   803→          const item = slashCommands[slash.index]
   804→          if (item) applyCommand(item)
   805→          else setSlash(null)
   806→          return
   807→        }
   808→        if (e.key === 'Escape') {
   809→          e.preventDefault()
   810→          setSlash(null)
   811→          return
   812→        }
   813→      }
   814→
   815→      const mod = e.metaKey || e.ctrlKey
   816→      if (mod && e.key.toLowerCase() === 'b') {
   817→        e.preventDefault()
   818→        wrapSelection('**', '**', 'bold text')
   819→        return
   820→      }
   821→      if (mod && e.key.toLowerCase() === 'i') {
   822→        e.preventDefault()
   823→        wrapSelection('*', '*', 'italic text')
   824→        return
   825→      }
   826→      if (mod && e.key.toLowerCase() === 'e') {
   827→        e.preventDefault()
   828→        wrapSelection('`', '`', 'code')
   829→        return
   830→      }
   831→      if (mod && e.key === 'Enter') {
   832→        e.preventDefault()
   833→        if (block.type === 'todo') toggleTodo(block.id)
   834→        else {
   835→          setActiveId(null)
   836→          ta.blur()
   837→        }
   838→        return
   839→      }
   840→
   841→      if (e.key === 'Enter' && !e.shiftKey) {
   842→        e.preventDefault()
   843→        // empty list/quote block → back to plain paragraph (Notion behaviour)
   844→        if (block.type !== 'p' && value.trim() === '') {
   845→          transformBlock(block.id, 'p', '')
   846→          return
   847→        }
   848→        const before = value.slice(0, s)
   849→        const after = value.slice(en)
   850→        const nextType: BlockType =
   851→          block.type === 'bullet' || block.type === 'numbered' || block.type === 'todo'
   852→            ? block.type
   853→            : 'p'
   854→        const nb = createBlock(nextType, after)
   855→        const arr = [...blocksRef.current]
   856→        const i = arr.findIndex((b) => b.id === block.id)
   857→        arr[i] = { ...arr[i], text: before }
   858→        arr.splice(i + 1, 0, nb)
   859→        commit(arr)
   860→        cancelBlurDeactivate()
   861→        setActiveId(nb.id)
   862→        focusBlock(nb.id, 'start')
   863→        return
   864→      }
   865→
   866→      if (e.key === 'Backspace' && s === en && s === 0) {
   867→        if (block.type !== 'p' && value === '') {
   868→          e.preventDefault()
   869→          transformBlock(block.id, 'p', '')
   870→          return
   871→        }
   872→        if (value === '') {
   873→          e.preventDefault()
   874→          const prev = idx > 0 ? blocksRef.current[idx - 1] : null
   875→          removeBlock(block.id)
   876→          if (prev && isTextual(prev.type)) activate(prev.id, 'end')
   877→          return
   878→        }
   879→        if (idx > 0) {
   880→          const prev = blocksRef.current[idx - 1]
   881→          if (prev.type === 'divider') {
   882→            e.preventDefault()
   883→            removeBlock(prev.id)
   884→            return
   885→          }
   886→          if (isMergeable(prev.type)) {
   887→            e.preventDefault()
   888→            const joinAt = prev.text.length
   889→            const arr = [...blocksRef.current]
   890→            arr[idx - 1] = { ...prev, text: prev.text + value }
   891→            arr.splice(idx, 1)
   892→            commit(arr)
   893→            cancelBlurDeactivate()
   894→            setActiveId(prev.id)
   895→            focusBlock(prev.id, joinAt)
   896→            return
   897→          }
   898→        }
   899→        return
   900→      }
   901→
   902→      if (e.key === 'ArrowUp' && s === en) {
   903→        const singleLine = ta.scrollHeight - ta.clientHeight < 4
   904→        if (s === 0 || singleLine) {
   905→          const prev = prevTextual(idx)
   906→          if (prev) {
   907→            e.preventDefault()
   908→            activate(prev.id, 'end')
   909→          }
   910→        }
   911→        return
   912→      }
   913→      if (e.key === 'ArrowDown' && s === en) {
   914→        const singleLine = ta.scrollHeight - ta.clientHeight < 4
   915→        if (s === value.length || singleLine) {
   916→          const nxt = nextTextual(idx)
   917→          if (nxt) {
   918→            e.preventDefault()
   919→            activate(nxt.id, 'start')
   920→          }
   921→        }
   922→        return
   923→      }
   924→      if (e.key === 'Tab') {
   925→        e.preventDefault()
   926→        const nxt = nextTextual(idx)
   927→        if (nxt) activate(nxt.id, 'start')
   928→        return
   929→      }
   930→      if (e.key === 'Escape') {
   931→        e.preventDefault()
   932→        setSlash(null)
   933→        setActiveId(null)
   934→        ta.blur()
   935→      }
   936→    }

  1496→    const startImageResize = (
  1497→      e: React.PointerEvent<HTMLElement>,
  1498→      block: Block,
  1499→      corner: 'nw' | 'ne' | 'sw' | 'se',
  1500→    ) => {
  1501→      e.preventDefault()
  1502→      e.stopPropagation()
  1503→      const handleEl = e.currentTarget
  1504→      const figure = handleEl.closest('figure')
  1505→      const baseW = figure?.getBoundingClientRect().width ?? 600
  1506→      const startX = e.clientX
  1507→      const startW = block.width ?? 100
  1508→      const dir = corner === 'ne' || corner === 'se' ? 1 : -1
  1509→      let finalW = startW
  1510→      setSelectedImageId(block.id)
  1511→      handleEl.setPointerCapture(e.pointerId)
  1512→      const onMove = (ev: PointerEvent) => {
  1513→        const pct = startW + ((ev.clientX - startX) / baseW) * 100 * dir
  1514→        finalW = Math.round(Math.min(100, Math.max(10, pct)))
  1515→        setImgResize({ id: block.id, w: finalW })
  1516→      }
  1517→      const finish = () => {
  1518→        handleEl.removeEventListener('pointermove', onMove)
  1519→        handleEl.removeEventListener('pointerup', finish)
  1520→        handleEl.removeEventListener('pointercancel', finish)
  1521→        setImgResize(null)
  1522→        setBlockImageWidth(block.id, finalW)
  1523→      }
  1524→      handleEl.addEventListener('pointermove', onMove)
  1525→      handleEl.addEventListener('pointerup', finish)
  1526→      handleEl.addEventListener('pointercancel', finish)
  1527→    }

  1646→    const handlePaste = (e: React.ClipboardEvent) => {
  1647→      const items = e.clipboardData?.items
  1648→      if (!items) return
  1649→      const files: File[] = []
  1650→      for (let i = 0; i < items.length; i++) {
  1651→        const it = items[i]
  1652→        if (it.kind === 'file' && it.type.startsWith('image/')) {
  1653→          const f = it.getAsFile()
  1654→          if (f) files.push(f)
  1655→        }
  1656→      }
  1657→      if (!files.length) return // normal text paste — let the browser handle it
  1658→      e.preventDefault()
  1659→      void insertImageFiles(files)
  1660→    }
  1661→
  1662→    const handleFileDragOver = (e: React.DragEvent) => {
  1663→      const types = Array.from(e.dataTransfer?.types ?? [])
  1664→      if (!types.includes('Files')) return // internal block reorder
  1665→      e.preventDefault()
  1666→      e.dataTransfer.dropEffect = 'copy'
  1667→      setFileDragOver(true)
  1668→    }
  1669→
  1670→    const handleFileDragLeave = (e: React.DragEvent) => {
  1671→      const related = e.relatedTarget as Node | null
  1672→      if (related && e.currentTarget.contains(related)) return
  1673→      setFileDragOver(false)
  1674→    }
  1675→
  1676→    const handleFileDrop = (e: React.DragEvent) => {
  1677→      setFileDragOver(false)
  1678→      const files = Array.from(e.dataTransfer?.files ?? []).filter((f) =>
  1679→        f.type.startsWith('image/'),
  1680→      )
  1681→      if (!files.length) return // not an image drop — let block reorder finish
  1682→      e.preventDefault()
  1683→      void insertImageFiles(files)
  1684→    }
  1685→
  1686→    const renderBlock = (block: Block, idx: number): ReactNode => {

  1688→      if (block.type === 'math') return renderMath(block, editing)
  1689→      if (block.type === 'code') return renderCode(block, editing)
  1690→      if (block.type === 'table') return renderTable(block, editing)
  1691→      if (block.type === 'image') return renderImage(block, idx)
  1692→      if (block.type === 'divider') return <hr className="border-border" />
  1693→      if (block.type === 'toc') return renderToc()
  1694→      return editing ? renderTextEditing(block, idx) : renderTextRendered(block, idx)
  1695→    }
